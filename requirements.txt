# Runtime dependencies.
#
# Every module imported under app/ has an entry here. Verified against a clean
# install (Phase 1 FR-2.1, FR-2.2, T-13).

# --- Web / API ---
fastapi==0.115.6
# Justified: the ASGI server. Not imported by app/ because it is the process
# entry point invoked as `uvicorn app.main:app`, not as a library.
uvicorn[standard]==0.34.0

# --- Telegram ---
aiogram==3.17.0

# --- Background tasks ---
celery==5.4.0
# Justified: no application module imports `redis` directly, but Celery's
# Redis broker transport requires it at runtime. Kept as an explicit pin so the
# broker dependency is visible rather than incidental.
redis==5.2.0

# --- Configuration ---
pydantic==2.10.6
pydantic-settings==2.7.0
# `python-dotenv` is deliberately not pinned: pydantic-settings declares it as
# a hard requirement, and the `env_file` support this project relies on comes
# from it.

# --- Outbound HTTP (Phase 3 P3.2) ---
# The audit's dependency audit (Phase 3 § "Dependency audit") called for httpx to
# be promoted from a transitive pin to an explicit one. It was previously
# present only because the openai SDK requires it and was never imported by
# app/. Phase 3 P3.2 imports it directly for the search provider adapter.
# httpx is pinned at the version already resolved transitively by
# openai==2.44.0, so this adds no new version to the tree.
httpx==0.28.1

# idna: IDNA (RFC 3490/5891) encoding of internationalised hostnames. The SSRF
# validator canonicalises a unicode hostname with it before resolution (P3.3).
# Imported lazily in app/infrastructure/fetch/ssrf.py, so only unicode
# hostnames need it at runtime, but it is declared explicitly because a clean
# install must be able to resolve it.
idna==3.18

# --- Content extraction (Phase 3 P3.4) ---
# trafilatura: readable-text extraction from fetched HTML (P3.4, FR-16). It is
# the extractor's main-document engine and is imported directly by
# app/infrastructure/extract/extractor.py, so it must be an explicit pin.
trafilatura==2.3.1

# --- AI provider (OpenAI-compatible) ---
# Added in Phase 1. Was imported by app/ but undeclared, so a clean environment
# could not start the application (audit C-2).
openai==2.44.0

# --- Data layer (Phase 2) ---
# The audit confirmed zero database code; Phase 2 introduces it. Both were
# declared-but-unused during Phase 1 and are now imported by app/db.
# asyncpg: the async PostgreSQL driver used by SQLAlchemy's async engine.
asyncpg==0.30.0
# sqlalchemy: the ORM and the migration target.
sqlalchemy==2.0.36
# alembic: schema migrations. Phase 2 requires a migration tool, and Phase 2 §6
# names the existing SQLAlchemy declaration as the natural target, so Alembic is
# the smallest solution that adds no second ORM or DDL framework.
alembic==1.14.0
# greenlet: a hard requirement of SQLAlchemy's asyncio support. Declared
# explicitly rather than left as an implicit transitive pin, because the
# application depends on it directly for every async database call.
greenlet==3.1.1
